using System;
using System.IO;
using System.Linq;
using System.Reflection;
using UnityEditor;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.SceneManagement;
using UnityEngine.UI;
using TMPro;
using Object=UnityEngine.Object;

[InitializeOnLoad]
public static class StageProgressSmoke
{
    const string Key="StageProgressSmoke.Running", Folder="/tmp/be-stage-progress-results";
    static int phase, frame, victorySettle;
    static double started;
    static StageSelectionController selection;
    static StageSelectionPopup popup;
    static UIManager ui;
    static StageProgress progress;
    static VoxelDefensePrototype game;
    static object Field(object o,string key)=>o.GetType().GetField(key,BindingFlags.NonPublic|BindingFlags.Instance).GetValue(o);
    static void Invoke(object o,string key)=>o.GetType().GetMethod(key,BindingFlags.NonPublic|BindingFlags.Instance).Invoke(o,null);
    static void Check(bool condition,string message){if(!condition)throw new Exception(message);}
    static void Pass(string message){Debug.Log("PROGRESS_PASS "+message);File.AppendAllText(Folder+"/playmode-checks.txt",message+"\n");}
    static Button ButtonNamed(string key)=>ui.GetComponentsInChildren<Button>(true).Single(x=>x.name==key && x.gameObject.activeInHierarchy);
    static void Click(string key)=>ButtonNamed(key).onClick.Invoke();
    static StageProgressSmoke(){EditorApplication.update+=Update;}
    public static void Run()
    {
        Directory.CreateDirectory(Folder);File.WriteAllText(Folder+"/playmode-checks.txt","");
        // The copied project's default prefix is replaced with a validation-only namespace before this run.
        Check(StageProgress.DefaultPrefix.Contains("Validation"),"Refuse to reset real progress");
        new StageProgress().Reset();
        EditorSceneManager.OpenScene("Assets/01_Scenes/02_Gameplay/GameScene.unity");
        SessionState.SetBool(Key,true);EditorApplication.EnterPlaymode();
    }
    static bool SelectionReady()
    {
        selection=Object.FindFirstObjectByType<StageSelectionController>();
        ui=Object.FindFirstObjectByType<UIManager>();
        popup=Object.FindFirstObjectByType<StageSelectionPopup>();
        if(selection==null||popup==null||!ui.HasOpenPopup||selection.CurrentStage!=null)return false;
        progress=(StageProgress)Field(selection,"progress");return progress!=null;
    }
    static void Update()
    {
        if(!SessionState.GetBool(Key,false)||!EditorApplication.isPlaying||EditorApplication.isCompiling)return;
        try
        {
            if(started==0)started=EditorApplication.timeSinceStartup;
            if(EditorApplication.timeSinceStartup-started>180)throw new Exception("Progress smoke timeout phase "+phase);
            if(++frame<5)return;frame=0;
            switch(phase)
            {
                case 0:
                    if(!SelectionReady())return;
                    Check(popup.UnlockedStages==1,"Fresh save must open only coast");
                    selection.SelectStage(4);Check(selection.SelectedStage==-1 && selection.CurrentStage==null,"Controller accepted locked stage");
                    Pass("Fresh start: one unlocked, no cleared cards; controller rejects locked capital.");
                    Capture("01-fresh");Click("progress.title");phase++;break;
                case 1:
                    Capture("02-management");Click("progress.reset");phase++;break;
                case 2:
                    Capture("03-reset-confirmation");Click("progress.cancel");Click("progress.close");
                    Check(popup.UnlockedStages==1,"Cancel changed progress");
                    selection.SelectStage(0);phase++;break;
                case 3:
                    if(selection.CurrentStage==null||Object.FindFirstObjectByType<StageTransition>(FindObjectsInactive.Include)?.IsPlaying==true)return;
                    game=Object.FindFirstObjectByType<VoxelDefensePrototype>();
                    FastFinish();phase++;break;
                case 4:
                    var victory=(VictoryPanelHud)Field(game,"victoryPanelHud");
                    if(!victory.IsOpen || ++victorySettle < 8)return;victorySettle=0;
                    Check(new StageProgress().IsCleared(1),"Victory did not save stage 1");
                    Pass("Stage 1 final-wave victory records clear through normal game Update.");
                    Capture("04-victory");
                    var next=(Button)Field(victory,"nextButton");Check(next.gameObject.activeInHierarchy,"Stage selection button hidden");
                    Check(((TextMeshProUGUI)Field(victory,"nextCaption")).text==GameText.Get("result.stage_select"),"Wrong victory button caption");
                    next.onClick.Invoke();phase++;break;
                case 5:
                    if(!SelectionReady())return;
                    Check(progress.IsCleared(1)&&popup.UnlockedStages==2,"Return lost clear/next stage");
                    var cards=(StageSelectionPopup.Card[])Field(popup,"cards");
                    Check(cards[0].chipText.text==GameText.Get("stage_select.state.cleared"),"Stage1 clear label wrong");
                    Check(cards[1].button.interactable&&!cards[2].button.interactable,"Next stage unlock wrong");
                    Pass("Victory selection return reloads saved clear; stage 2 unlocked, stage 3 locked.");
                    Capture("05-after-first-clear");selection.SelectStage(1);phase++;break;
                case 6:
                    if(selection.CurrentStage==null||Object.FindFirstObjectByType<StageTransition>(FindObjectsInactive.Include)?.IsPlaying==true)return;
                    game=Object.FindFirstObjectByType<VoxelDefensePrototype>();FastFinish();phase++;break;
                case 7:
                    var win=(VictoryPanelHud)Field(game,"victoryPanelHud");if(!win.IsOpen)return;
                    Check(new StageSpellUnlocks().MissingStage(DefenseBattle.Spell.MoveSpeedBuff)==0,"Stage2 clear did not unlock speed spell");
                    Pass("Stage 2 final-wave victory unlocks speed spell from the same saved clear record.");
                    ((Button)Field(win,"nextButton")).onClick.Invoke();phase++;break;
                case 8:
                    if(!SelectionReady())return;
                    Check(popup.UnlockedStages==3,"Stage3 not unlocked");
                    Click("progress.title");Click("progress.unlock_all");phase++;break;
                case 9:
                    Capture("06-unlock-confirmation");((Button)Field(Field(popup,"progressMenu"),"confirmButton")).onClick.Invoke();
                    Check(popup.UnlockedStages==5 && !progress.IsCleared(5),"Unlock all fabricated victory");
                    Pass("Unlock all opens five stages without fabricating clears or skill unlocks.");
                    Capture("07-all-unlocked");SceneManager.LoadScene("GameScene");phase++;break;
                case 10:
                    if(!SelectionReady())return;
                    Check(popup.UnlockedStages==5 && progress.IsCleared(2),"Scene reload lost unlock-all/save");
                    Check(new StageSpellUnlocks().MissingStage(DefenseBattle.Spell.ArmorBuff)==3,"Unlock all unlocked armor spell");
                    Click("progress.title");Click("progress.reset");Click("progress.cancel");Click("progress.close");
                    Check(popup.UnlockedStages==5 && progress.IsCleared(2),"Reset cancellation modified save");
                    Pass("Reload preserves unlock-all and clears; cancel reset preserves both.");
                    Click("progress.title");Click("progress.reset");Click("progress.reset");phase++;break;
                case 11:
                    Check(popup.UnlockedStages==1&&!progress.IsCleared(1)&&!progress.IsCleared(2),"Reset failed");
                    Check(new StageSpellUnlocks().MissingStage(DefenseBattle.Spell.MoveSpeedBuff)==2,"Reset did not relock spell");
                    Capture("08-reset");SceneManager.LoadScene("GameScene");phase++;break;
                case 12:
                    if(!SelectionReady())return;
                    Check(popup.UnlockedStages==1 && !progress.AllStagesUnlocked && !progress.IsCleared(2),"Reset did not persist");
                    Pass("Confirmed reset relocks stages and earned spells; reload preserves clean state.");
                    SessionState.SetBool(Key,false);Debug.Log("STAGE_PROGRESS_SMOKE_PASS");EditorApplication.Exit(0);break;
            }
        }
        catch(Exception e){Debug.LogException(e);SessionState.SetBool(Key,false);EditorApplication.Exit(1);}
    }
    static void FastFinish()
    {
        // Accelerated integration fixture only: remove enemies, leaving the final cleanup tick to actual Update.
        game.enabled=false;
        var battle=(DefenseBattle)Field(game,"battle");
        int limit=(int)Field(battle,"roundLimit");Check(limit>0,"No final wave");
        for(int i=0;i<100000;i++)
        {
            if(battle.CurrentPhase==DefenseBattle.Phase.Preparation)Invoke(game,"StartBattle");
            foreach(var enemy in battle.Enemies)enemy.Die();
            var spawned=(int[])Field(battle,"spawnedPerFaction");var totals=(int[])Field(battle,"roundSpawnCounts");
            if(battle.Round==limit && spawned.Zip(totals,(a,b)=>a>=b).All(x=>x))
            {game.enabled=true;return;}
            battle.Tick(DefenseBattle.TickDuration,false);
        }
        throw new Exception("Could not reach final wave");
    }
    static void Capture(string name)
    {
        var canvas=ui.GetComponent<Canvas>();var cam=Object.FindObjectsByType<Camera>(FindObjectsSortMode.None).First(x=>x.isActiveAndEnabled);
        int mask=cam.cullingMask;cam.cullingMask=-1;var mode=canvas.renderMode;var oldCam=canvas.worldCamera;float distance=canvas.planeDistance;
        var rt=new RenderTexture(1280,720,24);var old=cam.targetTexture;var active=RenderTexture.active;
        canvas.renderMode=RenderMode.ScreenSpaceCamera;canvas.worldCamera=cam;canvas.planeDistance=1;
        cam.targetTexture=rt;Canvas.ForceUpdateCanvases();cam.Render();RenderTexture.active=rt;
        var tex=new Texture2D(1280,720,TextureFormat.RGB24,false);tex.ReadPixels(new Rect(0,0,1280,720),0,0);tex.Apply();
        File.WriteAllBytes(Folder+"/"+name+".png",tex.EncodeToPNG());
        cam.cullingMask=mask;cam.targetTexture=old;RenderTexture.active=active;canvas.renderMode=mode;canvas.worldCamera=oldCam;canvas.planeDistance=distance;
        Object.DestroyImmediate(tex);rt.Release();Object.DestroyImmediate(rt);
    }
}
